## Usage

```tsx
import {
  Accordion,
  AccordionContent,
  AccordionItem,
  AccordionTrigger,
} from "@/components/ui/accordion";
```

```tsx
<Accordion defaultValue={["details"]}>
  <AccordionItem value="details">
    <AccordionTrigger autoFocus label="Details" />
    <AccordionContent>
      <text>Expanded content.</text>
    </AccordionContent>
  </AccordionItem>
</Accordion>
```

Triggers support Enter, Space, pointer activation, Up/Down focus movement, and
Home/End. Add `multiple` to allow more than one item to remain open.
