## Usage

```tsx
import { PasswordInput } from "@/components/ui/password-input";
```

```tsx
<PasswordInput
  label="Password"
  placeholder="Enter password"
  showToggle
  autoFocus
  onSubmit={(value) => console.log("Password:", value)}
/>
```
